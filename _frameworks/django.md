```bash
# install with pip
pip install django
```

## 3 Concepts

- Each Django project consists of one or more apps
  - Each app encapsulates related functionality
  - Apps can exist independently of a project and be incorporated into projects
    as needed

## 4 Usage

### 4.1 Django CLI

- Django can be managed with the Django CLI

```bash
# create a Django project with the specified name in the current directory
django-admin startproject myproject

# create a Django project with the specified name in the specified directory
django-admin startproject myproject ./mydirectory
```

### 4.2 Management Script

- Existing Django projects can be managed with the `manage.py` script inside the project's
  root directory

```bash
# create a Django app with the specified name in the current Django project
python manage.py startapp myapp

# start the local development server for the current Django project on port 8000
python manage.py runserver

# generate migration files from the models.py files of the Django apps
python manage.py makemigrations

# apply the migration files generated by the makemigrations command
python manage.py migrate

# create a superuser for the current Django project (requires migrations beforehand)
python manage.py createsuperuser

# open a Python shell with the Django project's settings and dependencies loaded
python manage.py shell

# run tests for the specified Django app
python manage.py test myapp
```

## 5 Project Structure

## 7 Views

- Views are functions or classes that process HTTP requests and return responses

```python
from django.http import HttpRequest, HttpResponse
from django.shortcuts import render
from django.views import View

from .forms import NameForm


# define a function-based view
def index(request: HttpRequest) -> HttpResponse:
    # define the response to the request
    return HttpResponse(content="Success!")


# define a class-based view
class Hello(View):
    # define a method for the class-based view
    def get(self, request: HttpRequest) -> HttpResponse:
        # define the response to the request
        return HttpResponse(content="Hello, World!")
```

### 7.1 HTTP Methods

```python
from django.http import HttpRequest, HttpResponse
from django.views import View


def index(request: HttpRequest) -> HttpResponse:
    # handle GET requests
    if request.method == "GET":
        # access a GET request parameter
        if "name" in request.GET:
            print(request.GET["name"])

        return HttpResponse(content="Hello, World!")

    # handle POST requests
    if request.method == "POST":
        # access a POST request parameter
        if "name" in request.POST:
            print(request.POST["name"])

        return HttpResponse(content="Success!")


class Hello(View):

    # define the method for GET requests to the view
    def get(self, request: HttpRequest) -> HttpResponse:
        # access a GET request parameter
        if "name" in request.GET:
            print(request.GET["name"])

        return HttpResponse(content="Hello, World!")

    # define the method for POST requests to the view
    def post(self, request: HttpRequest) -> HttpResponse:
        # access a POST request parameter
        if "name" in request.POST:
            print(request.POST["name"])

        return HttpResponse(content="Success!")
```

### 7.2 URL Parameters

```python
from django.http import HttpRequest, HttpResponse

def greet(request: HttpRequest, url_parameter: int, other_url_parameter: str) -> HttpResponse:
```

### 7.3 Redirects

```python
from django.http import HttpRequest, HttpResponse, HttpResponseRedirect
from django.shortcuts import redirect
from django.urls import reverse


def redirect1(request: HttpRequest) -> HttpResponse:
    # return a redirect to the specified URL
    return HttpResponseRedirect(
        redirect_to=reverse(         # dynamically create URL
            viewname="myapp:index",  # URL pattern name (can be namespaced)
            args=[4, "John"],        # URL parameters to pass
        ),
    )


def redirect2(request: HttpRequest) -> HttpResponse:
    # return a redirect to the specified URL and pass according URL parameters
    return redirect(
        "myapp:index",  # URL pattern name (can be namespaced)
        4, "John",      # URL parameters to pass
    )
```

### 7.4 Templates

```python
from django.http import HttpRequest, HttpResponse
from django.shortcuts import render
from django.template import loader


def index(request: HttpRequest) -> HttpResponse:
    # render a template as the response
    template = loader.get_template("polls/index.html")  # template to render
    return HttpResponse(
        content=template.render(       # render template
            context={"name": "John"},  # pass values to insert into template
            request=request,           # pass original request
        ),
    )


def home(request: HttpRequest) -> HttpResponse:
    # shortcut for returning a rendered template
    return render(
        request=request,             # pass original request
        template_name="index.html",  # pass template to render
        context={"name": "John"},    # pass values to insert into template
    )
```

### 7.5 Errors

```python
from django.http import Http404, HttpRequest, HttpResponse
from django.shortcuts import get_list_or_404, get_object_or_404, render

from .models import MyModel  # custom model


def error(request: HttpRequest) -> HttpResponse:
    try:
        MyModel.objects.get(pk=0)
    # catch the exception raised when the model entity cannot be found
    except MyModel.DoesNotExist:
        # raise an HTTP 404 error for the request
        raise Http404("This object does not exist")

    return HttpResponse("Success!")


def search(request: HttpRequest) -> HttpResponse:
    # retrieve the first matching model entity or raise an HTTP 404 error
    model = get_object_or_404(klass=MyModel, pk=0)

    # retrieve thematching model entities or raise an HTTP 404 error
    models = get_list_or_404(klass=MyModel, text="Hi")

    return HttpResponse(model.text)
```

### 7.6 Forms

```python
from django.http import HttpRequest, HttpResponse

from .forms import NameForm  # custom form


def greet(request: HttpRequest) -> HttpResponse:
    if request.method == "POST":

        # populate the form with POST data
        form: NameForm = NameForm(data=request.POST)
```

### 7.7 Generic Views

- Generic views provide reusable views for common model-related operations
  - They are configured with specific models, fields, and methods
  - Their URL mappings must therefore be configured accordingly
  - Their templates must therefore follow the expected structure

#### 7.7.1 Read-Only

```python
from django.db.models import Model, QuerySet
from django.views import generic

from .models import MyModel


# define a generic view for displaying a single model entity
class DetailView(generic.DetailView):
    # define the model for the view
    model: Model = MyModel

    # define the template to render
    template_name: str = "polls/detail.html"

    # define the context variable name used in the template
    context_object_name: str = "model"


# define a generic view for displaying a list of model entities
class IndexView(generic.ListView):
    # define the template to render
    template_name: str = "index.html"

    # define the context variable name used in the template
    context_object_name: str = "latest_question_list"

    # return the query set used to generate the view
    def get_queryset(self) -> QuerySet[MyModel]:
        return MyModel.objects.order_by("-pub_date")[:5]

```

## 8 Routing

- URL patterns map specific URLs to defined views
  - The mapped views are called when the specified URLs are requested

- URL patterns are defined in the `urls.py` files of the Django project and its apps

- The project's `urls.py` file maps URL prefixes to Django apps

```python
from django.contrib import admin
from django.urls import URLPattern, include, path


# map URL prefixes to app URLs (checked in order)
urlpatterns: list[URLPattern] = [
    # map the URL prefix to the built-in admin site
    path(route="admin/", view=admin.site.urls),

    # map the URL prefix to all URL patterns of the specified app
    path(route="", view=include(arg="myapp.urls")),
]
```

- An app's `urls.py` file maps URLs to its views
  - This `urls.py` file must be created manually

```python
from django.urls import URLPattern, path

from . import views


# define the namespace for URL patterns of the Django app
app_name: str = "demo"

# map URLs to views (checked in order)
urlpatterns: list[URLPattern] = [
    # map a URL to a view function
    path(
        route="hello",     # URL path
        view=views.hello,  # function-based view
        name="hello",      # internal URL pattern name
    ),

    # map a URL to a class-based or generic view
    path(
        route="hi",               # URL path
        view=views.Hi.as_view(),  # class-based or generic view
        name="hi",                # internal URL pattern name
    ),

    # define URL parameters that are passed to its mapped view as arguments
    path(
        route="path/<int:some_id>/to/<str:some_name>",  # define parameter types
        view=views.hey,
        name="hey",
    ),
]
```

## 9 Models

- Models are Python representations of database tables managed by Django
  - Database tables are derived from them through migrations

- Models are defined in the `models.py` file of the corresponding Django app

- To apply model changes to the database, migrations must be created and applied
  (see [4 Usage](#4-usage))

```python
from django.contrib import admin
from django.db.models import (
    CharField,
    DateTimeField,
    ForeignKey,
    IntegerField,
    Model,
)
from django.utils import timezone


# define a model with an automatically generated primary key
class MyModel(Model):
    # define a text field
    text: CharField = CharField(
        name="Text",     # human-readable name used by Django
        max_length=255,  # limit the length of the text
        default="???",   # default value of field
    )

    # define an integer field
    score: IntegerField = IntegerField(
        name="Score",  # human-readable name used by Django
        default=0,     # default value of field
    )

    # define a date field
    created_at: DateTimeField = DateTimeField(
        name="Created at",      # human-readable name used by Django
        auto_now=False,         # whether to populate with the current date time
        default=timezone.now,  # default value of the field
    )

    # define the string representation used when the object is displayed as text
    def __str__(self) -> str:
        return self.text

    # configure how the method is displayed in the admin site
    @admin.display(
        boolean=True,
        ordering="text",
        description="Does stuff?",
    )
    def do_stuff(self) -> bool:
        return True
```

### 9.1 Relationships

```python
from django.db import models
from django.db.models import (
    CharField,
    ForeignKey,
    ManyToManyField,
    Model,
)
from django.utils import timezone


class Question(Model):
    text: CharField = CharField(max_length=255)

    def __str__(self) -> str:
        return self.text


class Choice(Model):
    # create a many-to-one relationship to another model
    question: ForeignKey = ForeignKey(
        to=Question,               # model to relate to
        on_delete=models.CASCADE,  # behavior when the related entity is deleted
    )

    text: CharField = CharField(max_length=255)


class Participant(Model):
    # create a many-to-many relationship to another model
    questions: ManyToManyField = ManyToManyField(
        to=Question,               # model to relate to
        on_delete=models.CASCADE,  # behavior when the related entity is deleted
    )

    name: CharField = CharField(max_length=255)
```

### 9.2 Entities

- Models can be used to access and manipulate their corresponding database entries

```python
from django.db.models import F
from django.db.models import QuerySet

from .models import MyModel


# query the database for entities of the specified model
result1: QuerySet[MyModel] = MyModel.objects.all()  # all entities as a query set
result2: MyModel = MyModel.objects.get(pk=1)        # entity with the specified primary key

# manipulate the query set
result1 = result1.order_by(  # order the query set
    "name",  # order by field in ascending order
    "-age",  # order by field in descending order
)

# create a new model entity
model: MyModel = MyModel(name="John")

# modify the model entity
model.score = F("score") + 1  # increment the score in the database

# save the model entity to the database
model.save()

# access fields of the entity
id: int = model.id  # requires the entity to be saved in the database
name: str = model.name
```

## 11 Templates

- Templates can be defined as HTML files in the `templates` directory of the corresponding
  Django app
  - This directory must be created manually

- Django uses its own templating language, the Django Template Language (DTL), to render
  templates
  - This allows dynamic insertion of values and generation of HTML at runtime

```html
<!DOCTYPE html>

<html lang="en">

    <head>
        <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <title>Document</title>
    </head>

    <body>
        <!-- insert value from view into template -->
        <p>Hello, {{ name }}!</p>

        <!-- insert a dictionary element from the view into the template -->
        <p>Hello, {{ person.name }}!</p>

        <!-- iterate over a collection from the view and generate HTML for each item -->
        {% for my_item in my_inserted_object %}
            <!-- access the current iteration item -->
            <p>{{ my_item.name }}</p>
        {% endfor %}

        <!-- insert URLs by their internal mapping names -->
        <a href="{% url 'index' %}">Index page</a>
        <a href="{% url 'myapp:about' %}">About us</a>. <!-- specify namespace of URL pattern -->
        <a href="{% url 'details' item.id %}">{{ item.name }}</a>  <!-- pass a URL parameter -->

        <!-- define form in template -->
        <form method="POST">
            {% csrf_token %}  <!-- insert the CSRF token for the form -->
            {{ form }}        <!-- render the form defined in forms.py -->
            <button type="submit">Save</button>
        </form>
    </body>

</html>
```

- Django templates can define blocks that can be overridden by child templates

```html
<!DOCTYPE html>

<html lang="en">

    <head>
        <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <title>Document</title>
    </head>

    <body>
        <!-- define a block that child templates can override -->
        {% block my_insertion %}
        {% endblock %}
    </body>

</html>
```

- Django templates can extend other templates

```html
<!-- extend the base template -->
{% extends "myapp/base.html" %}

<!-- override a block of the base template -->
{% block my_insertion %}
<p>
    This is dynamically inserted HTML!
</p>
{% endblock %}
```

- <u>Best practices</u>:
  - Template files in the `templates` directory should be namespaced by placing them inside
    an additional directory with the same name as the corresponding Django app

## 13 Tests

- Django's test discovery finds test files whose names match the `test*.py` pattern

### 13.1 Unit Tests

```python
from django.test import TestCase

from .exceptions import MyException
from .models import MyModel


# create a test class
class MyModelTests(TestCase):
    # define a test method
    def test_all(self):
        model: MyModel = MyModel()

        # assert equality
        self.assertEqual(
            first=model.get_text(),            # first expression
            second="Hello, World!",            # second expression
            msg="The values were not equal!",  # optional failure message
        )

        # assert approximate equality
        self.assertAlmostEqual(
            first=model.get_price(),           # first real number
            second=3.14,                       # second real number
            places=2,                          # number of decimal places to compare (optional)
            msg="The values were not equal!",  # optional failure message
        )

        # assert truth
        self.assertTrue(
            expr=model.is_active(),         # expression to check
            msg="The value was not true!",  # optional failure message
        )

        # assert falsehood
        self.assertFalse(
            expr=model.is_deleted(),         # expression to check
            msg="The value was not false!",  # optional failure message
        )

        # assert that an exception is raised
        self.assertRaises(
            expected_exception=MyException,  # expected exception
            callable=lambda: model.panic(),  # callable that should raise the exception
        )

    # define another test method
    def test_other(self):
        model: MyModel = MyModel()
        self.assertTrue(expr=model.is_active(), msg="This wasn't equal!")
```

### 13.2 Endpoint Tests

```python
from django.http import HttpResponse
from django.test import TestCase


class MyViewTest(TestCase):
    def test_get_text(self):
        # simulate an HTTP request
        response: HttpResponse = self.client.get(

```

## 14 Administration

- Django provides an admin site by default for managing users, permission groups, and
  registered models
  - This panel is reachable at `/admin/`
  - To access the panel, at least one superuser must be created, and migrations must have been
    performed (see [4 Usage](#4-usage))

- Models must be registered in the `admin.py` file of the corresponding Django app to make them
  accessible in the admin panel
  - There they can also be configured for the admin site

```python
from django.contrib import admin
from django.db.models import Model

from .models import Choice, Question


# define an inline class that enables related models to be edited in the admin site
class ChoiceStackedInline(admin.StackedInline):  # stacked view
    model: Model = Choice  # which model to add
    extra: int = 4         # how many create slots should be provided per default (optional)


# define an inline class that enables related models to be edited in the admin site
class ChoiceTabularInline(admin.TabularInline):  # tabular view
    model: Model = Choice  # which model to add
    extra: int = 4         # how many create slots should be provided per default (optional)


# define an admin class that controls how a model is displayed in the admin site (optional)
class QuestionAdmin(admin.ModelAdmin):
    # define which model fields to display when creating or editing a model
    fieldsets = [
        # fieldset without a heading
        (None, {"fields": ["text"]}),

        # fieldset with a heading
        ("Date information", {
            "fields": ["pub_date"],   # fields to display in the fieldset
            "classes": ["collapse"],  # whether the fieldset starts collapsed (optional)
        }),
    ]

    # define which inline models should be included
    inlines: list[type] = [ChoiceTabularInline]

    # define which model fields to display in the model list view
    list_display: list[str] = ["question_text", "pub_date"]

    # add filter options for the specified fields in the model list view
    list_filter: list[str] = ["pub_date"]

    # add search capability for the specified fields in the model list view
    search_fields: list[str] = ["text"]


# register the model in the admin site
admin.site.register(
    model_or_iterable=Question,  # model to register
    admin_class=QuestionAdmin,   # optional admin class to use for the model
)
```
